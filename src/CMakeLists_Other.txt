set (CMAKE_MODULE_PATH "../cmake;${CMAKE_MODULE_PATH}")
find_package(HDF5 REQUIRED COMPONENTS C)
find_package(OpenMP REQUIRED)

set(ARCH "native" CACHE STRING "architecture to use for -march flag to compiler")

set(DEBUG_FLAGS "-g")

set (OMP_FLAGS "${OpenMP_CXX_FLAGS}")
set (DEFINES "-DR123_NO_SINCOS -Drestrict=__restrict__ -DNDEBUG")
set (OPT     "-O3 -ffast-math -fno-finite-math-only -fno-omit-frame-pointer -march=${ARCH}")
set (WARN    "-pedantic -Wall -Wno-strict-overflow -Wno-unused-function -Wno-unknown-pragmas")

set (CMAKE_CXX_FLAGS "${OMP_FLAGS} ${CMAKE_CXX_FLAGS} ${DEBUG_FLAGS} -std=c++11 ${DEFINES} ${OPT} ${WARN}" )
include_directories(SYSTEM "include")

set (ENGINE_SRC 
    main.cpp
    martini.cpp
    martini_fix_rigid.cpp
    martini_hybrid.cpp
    martini_masses.cpp
    martini_potential.cpp
    martini_stage_params.cpp
    martini_brownian.cpp
    box.cpp
    environment.cpp
    hbond.cpp 
    rotamer.cpp
    placement.cpp
    rama_map_pot.cpp
    coord_operator.cpp
    coord_basic.cpp
    coord_const.cpp
    coord_map.cpp
    spline.cpp
    spring.cpp
    tension.cpp
    sigmoid_energy.cpp
    deriv_engine.cpp 
    sidechain_radial.cpp
    backbone_steric.cpp 
    eig.cpp 
    membrane_potential.cpp
    surface.cpp
    timing.cpp 
    thermostat.cpp
    h5_support.cpp 
    state_logger.cpp
    monte_carlo_sampler.cpp
    engine_c_library.cpp)

# The device engine (device_engine.h).  Without CUDA its files compile as C++ and run on the host.
set(DEVICE_SRC
    device.cu
    device_engine.cu
    device_coord.cu
    device_spring.cu
    device_hbond.cu
    device_rama.cu
    device_affine.cu
    device_placement.cu
    device_environment.cu
    device_rotamer.cu
    device_martini.cu)
option(UPSIDE_CUDA "compile the device files for NVIDIA GPUs" OFF)
if(UPSIDE_CUDA)
    # one binary for Kepler (the K80, which needs CUDA 11) to Ampere; it links the CUDA runtime statically, so it
    # still loads on nodes without a GPU
    cmake_minimum_required(VERSION 3.18)
    set(CMAKE_CUDA_ARCHITECTURES "37;60;70;75;80;86" CACHE STRING "GPU architectures to compile for")
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 14)
    set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} --extended-lambda -Wno-deprecated-gpu-targets ${DEFINES} -Xcompiler=-O3,-ffast-math,-fno-finite-math-only,-march=${ARCH}")
else()
    set_source_files_properties(${DEVICE_SRC} PROPERTIES LANGUAGE CXX COMPILE_FLAGS "-x c++")
endif()
list(APPEND ENGINE_SRC ${DEVICE_SRC})

add_executable (upside ${ENGINE_SRC})

INCLUDE_DIRECTORIES (${HDF5_INCLUDE_DIRS})
target_link_libraries(upside stdc++ ${HDF5_LIBRARIES})

find_package(Eigen3 REQUIRED)
include_directories(SYSTEM ${EIGEN3_INCLUDE_DIR})

add_library(upside_calculation SHARED ${ENGINE_SRC})

set_target_properties(upside_calculation PROPERTIES
    COMPILE_FLAGS "-DPARAM_DERIV"
    OUTPUT_NAME   "upside")

target_link_libraries(upside_calculation stdc++ ${HDF5_LIBRARIES})
